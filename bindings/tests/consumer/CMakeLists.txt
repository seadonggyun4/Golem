cmake_minimum_required(VERSION 3.21)
project(GolemBindingConsumer LANGUAGES C)
find_package(Golem 0.1.0 EXACT CONFIG REQUIRED)
if(NOT TARGET Golem::binding)
    message(FATAL_ERROR "Install Golem with GOLEM_BUILD_BINDINGS=ON")
endif()
add_executable(binding_consumer main.c)
target_link_libraries(binding_consumer PRIVATE Golem::binding)
enable_testing()
add_test(NAME installed_binding COMMAND binding_consumer)
