set(fuzz_names adapter_json adapter_msgpack journal parser policy_artifact)
add_executable(golem_fuzz_seeds seeds.c)
target_link_libraries(golem_fuzz_seeds PRIVATE Golem::golem)
golem_configure_target(golem_fuzz_seeds)
add_test(NAME fuzz_seeds COMMAND golem_fuzz_seeds
    "${CMAKE_CURRENT_BINARY_DIR}/seeds" "${PROJECT_SOURCE_DIR}")
set_tests_properties(fuzz_seeds PROPERTIES FIXTURES_SETUP fuzz_corpus LABELS fuzz TIMEOUT 30)

foreach(name IN LISTS fuzz_names)
    set(source "${name}.c")
    if(name STREQUAL "adapter_msgpack")
        set(source adapter_json.c)
    endif()
    if(BUILD_TESTING)
        add_executable(golem_mutate_${name} ${source} smoke.c)
        target_link_libraries(golem_mutate_${name} PRIVATE Golem::golem)
        golem_configure_target(golem_mutate_${name})
        if(name STREQUAL "adapter_msgpack")
            target_compile_definitions(golem_mutate_${name} PRIVATE GOLEM_FUZZ_MSGPACK=1)
        endif()
        add_test(NAME mutation_${name} COMMAND golem_mutate_${name}
            "${CMAKE_CURRENT_BINARY_DIR}/seeds/${name}")
        set_tests_properties(mutation_${name} PROPERTIES FIXTURES_REQUIRED fuzz_corpus
            LABELS hardening TIMEOUT 120)
    endif()
    if(GOLEM_BUILD_FUZZERS)
        add_executable(golem_fuzz_${name} ${source})
        target_link_libraries(golem_fuzz_${name} PRIVATE Golem::golem)
        if(GOLEM_LIBFUZZER_LIBRARY)
            target_compile_options(golem_fuzz_${name} PRIVATE -fsanitize=fuzzer-no-link)
            target_link_libraries(golem_fuzz_${name} PRIVATE "${GOLEM_LIBFUZZER_LIBRARY}" c++)
        else()
            target_compile_options(golem_fuzz_${name} PRIVATE -fsanitize=fuzzer)
            target_link_options(golem_fuzz_${name} PRIVATE -fsanitize=fuzzer)
        endif()
        golem_configure_target(golem_fuzz_${name})
        if(name STREQUAL "adapter_msgpack")
            target_compile_definitions(golem_fuzz_${name} PRIVATE GOLEM_FUZZ_MSGPACK=1)
        endif()
        set(max_len 16385)
        if(name STREQUAL "journal" OR name STREQUAL "parser")
            set(max_len 1048609)
        elseif(name STREQUAL "policy_artifact")
            set(max_len 33)
        elseif(name STREQUAL "adapter_msgpack")
            set(max_len 4097)
        endif()
        file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/findings/${name}")
        file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/corpus/${name}")
        add_test(NAME fuzz_${name} COMMAND golem_fuzz_${name}
            -runs=10000 -seed=1701 -max_len=${max_len} -timeout=10 -rss_limit_mb=1024
            "-artifact_prefix=${CMAKE_CURRENT_BINARY_DIR}/findings/${name}/"
            "${CMAKE_CURRENT_BINARY_DIR}/corpus/${name}"
            "${CMAKE_CURRENT_BINARY_DIR}/seeds/${name}")
        set_tests_properties(fuzz_${name} PROPERTIES FIXTURES_REQUIRED fuzz_corpus LABELS fuzz
            TIMEOUT 180 ENVIRONMENT "ASAN_OPTIONS=halt_on_error=1;UBSAN_OPTIONS=halt_on_error=1:print_stacktrace=1")
    endif()
endforeach()
